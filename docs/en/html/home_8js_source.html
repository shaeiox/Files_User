293 // Triggered on Dropdown selections
294 function applyFiltersAndRender() {
295 let filtered = [...rawResources];
296
297 // 1. Apply Category Filter
298 if (activeCategory !== "all") {
299 filtered = filtered.filter(item => item.id === activeCategory);
300 }
301
302 // 2. Apply Sorting Filter
303 filtered.sort((a, b) => {
304 if (activeSort === "name-asc") {
305 const titleA = a.name || "";
306 const titleB = b.name || "";
307 return titleA.localeCompare(titleB);
308 } else if (activeSort === "count-desc") {
309 const countA = a.stats?.total || 0;
310 const countB = b.stats?.total || 0;
311 return countB - countA;
312 } else { // "latest"
313 const timeA = a.stats?.latestModified || 0;
314 const timeB = b.stats?.latestModified || 0;
315 return timeB - timeA;
316 }
317 });
318
319 renderResourceCards(filtered);
320 }

322 function renderResourceCards(resources) {
323 const d = typeof window.getDict === 'function' ? window.getDict() : getDict();
324 const grid = document.getElementById("resourcesGrid");
325 if (!grid) return;
326
327 if (resources.length === 0) {
328 grid.innerHTML = `<div class="ac-empty" style="grid-column:1/-1; padding:32px; text-align:center;">${d.empty || "No resources found."}</div>`;
329 return;
330 }
331
332 grid.innerHTML = resources.map(resource => {
333 const title = resource.name;
334 const desc = d.categoryDesc || "Browse and download files in this category.";
335
336 let statsBadge;
337 if (!resource.stats || !resource.stats.available) {
338 statsBadge = `<span class="badge badge-error">${d.unavailable || "Unavailable"}</span>`;
339 } else if (resource.stats.total === 0) {
340 statsBadge = `<span class="stat-empty">${d.emptyFolder || "Empty"}</span>`;
341 } else {
342 const fileCount = resource.stats.files || 0;
343 const folderCount = resource.stats.folders || 0;
344 const parts = [];
345 if (fileCount) parts.push(fileCount + " " + (fileCount === 1 ? (d.file || "file") : (d.files || "files")));
346 if (folderCount) parts.push(folderCount + " " + (folderCount === 1 ? (d.folder || "folder") : (d.folders || "folders")));
347 statsBadge = parts.join(" · ") || (d.emptyFolder || "Empty");
348 }
349
350 let updatedLabel = d.updated || "Updated";
351 if (resource.stats && resource.stats.latestModified) {
352 const lang = typeof window.getLang === 'function' ? window.getLang() : getLang();
353 updatedLabel = formatDate(resource.stats.latestModified, lang);
354 }
355
356 const statusHtml = (resource.stats && resource.stats.available)
357 ? `<span class="status-live"><span class="dot"></span> ${updatedLabel}</span>`
358 : `<span class="status-error"><span class="dot dot-error"></span> ${d.offline || "Offline"}</span>`;
359
360 return resourceCard(resource.path, title, desc, statsBadge, resource.iconType, statusHtml);
361 }).join('');
362
363 updateLastSyncTime();
364 }

452 function humanSize(bytes) {
453 if (!Number.isFinite(bytes) || bytes < 0) return "—";
454 if (bytes === 0) return "0 B";
455 const units = ["B", "KB", "MB", "GB", "TB"];
456 const i = Math.min(Math.floor(Math.log(bytes) / Math.log(1024)), units.length - 1);
457 return (bytes / Math.pow(1024, i)).toFixed(i === 0 ? 0 : 1) + " " + units[i];
458 }

710 async function scanAll(query) {
711 const d = typeof window.getDict === 'function' ? window.getDict() : getDict();
712 stopFlag = false;
713
714 const resultsBox = document.getElementById("resultsBox");
715 const resultsList = document.getElementById("resultsList");
716 const progressEl = document.getElementById("progress");
717 const searchBtn = document.getElementById("searchBtn");
718 const stopBtn = document.getElementById("stopBtn");
719 const qInput = document.getElementById("globalSearch");
720
721 if (!resultsBox) return;
722
723 resultsList.innerHTML = "";
724 resultsBox.style.display = "block";
725 resultsBox.scrollIntoView({ behavior: "smooth", block: "nearest" });
726 progressEl.textContent = d.scanning || "Scanning…";
727 stopBtn.style.display = "inline-flex";
728 searchBtn.disabled = true;
729 qInput.disabled = true;
730
731 const q = query.trim();
732 let found = 0;
733 let scanned = 0;
734
735 try {
736 const data = await ResourceService.search(q, { limit: SEARCH_MAX_RESULTS });
737 scanned = data.scanned || 0;
738 const entries = data.results || [];
739 for (const e of entries) {
740 if (stopFlag) break;
741 if (!e || !e.name) continue;
742 const type = e.type === "directory" ? "directory" : "file";
743 const href = type === "directory"
744 ? ResourceService.browseHref(e.path)
745 : ResourceService.downloadHref(e.path);
746 addResult({ name: e.name, href, path: e.path, type });
747 found++;
748 }
749 } catch { }
750
751 progressEl.textContent = `${d.scannedDirs || 'Scanned'} ${scanned} · ${found} ${d.foundText || 'found'}`;
752
753 progressEl.textContent = `${d.done || 'Done'} · ${found} ${d.foundText || 'found'}`;
754 stopBtn.style.display = "none";
755 searchBtn.disabled = false;
756 qInput.disabled = false;
757
758 if (!found) {
759 resultsList.innerHTML = `<div class="ac-empty">${d.emptySearch || 'No results for'} "${query}"</div>`;
760 }
761 }
