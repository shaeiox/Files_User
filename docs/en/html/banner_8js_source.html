380 function getSessionStart() {

420 function dismiss(el) {
421 if (!el || el.classList.contains("cav-out")) return;
422 el.classList.add("cav-out");

424 el.remove();

502 function reset() {
503 try {
504 localStorage.removeItem(CONFIG.shownKey);
505 sessionStorage.removeItem(CONFIG.startKey);
506 log("All storage cleared — banner will re-trigger");
507 } catch { }
508 }

510 /* ═════════════════════════════════════════════
511 ⏱️ AUTO TRIGGER (time-based, cross-page)
512 Uses sessionStorage to persist start time
513 across page navigations.
514 ═════════════════════════════════════════════ */
515 let triggerTimer = null;
516 let triggerFired = false;
517
518 function startTrigger() {
519 if (triggerFired) return;
520
521 // Check if already shown before (production mode)
522 if (!CONFIG.debugMode && wasShownBefore()) {
523 log("Trigger skipped — banner was already shown");
524 return;
525 }
526
527 const sessionStart = getSessionStart();
528 const elapsed = Date.now() - sessionStart;
529 const remaining = Math.max(0, CONFIG.delay - elapsed);
530
531 if (remaining <= 0) {
532 // Already past the delay threshold (user navigated between pages)
533 fireTrigger();
534 } else {
535 log(`Trigger set — banner in ${(remaining / 1000).toFixed(1)}s (${(elapsed / 1000).toFixed(1)}s already elapsed)`);
536 triggerTimer = setTimeout(fireTrigger, remaining);
537 }
538 }
539
540 function fireTrigger() {
541 if (triggerFired) return;
542 triggerFired = true;
543
544 if (triggerTimer) {
545 clearTimeout(triggerTimer);
546 triggerTimer = null;
547 }
548
549 const elapsed = (getSessionTime() / 1000).toFixed(1);
550 log(`✨ Trigger fired after ${elapsed}s total session time`);
551
552 showOnce();
553 }

561 log("Booted on", window.location.pathname);
562 }
563
564 if (document.readyState === "loading") {
565 document.addEventListener("DOMContentLoaded", boot);
566 }
