136 async function fetchList() {
137 const d = typeof window.getDict === 'function' ? window.getDict() : getDict();
138 try {
139 const relPath = currentPath.replace(/^\/browse\/?/, "");
140 const data = await ResourceService.listFiles(relPath);
141 if (!Array.isArray(data)) throw new Error("Bad JSON");
142
143 allFiles = data
144 .filter(x => x && x.name && x.name !== "." && x.name !== "..")
145 .filter(x => !isHidden(x.name))
146 .map(x => ({
147 name: String(x.name),
148 relPath: x.path || (relPath ? relPath.replace(/\/+$/, "") + "/" + x.name : x.name),
149 type: x.type === "directory" ? "directory" : "file",
150 size: Number(x.size || 0),
151 modified: x.mtime || x.modified || x.time || 0

163 /* ═══════════════════════════════════════════════
164 DATA PIPELINE (Filter -> Search -> Sort -> Render)
165 ═══════════════════════════════════════════════ */
166 function applyFiltersAndRender() {
167 const d = typeof window.getDict === 'function' ? window.getDict() : getDict();
168 let filtered = [...allFiles];
169
170 // 1. Apply Search Query
171 if (searchQuery) {
172 filtered = filtered.filter(f => f.name.toLowerCase().includes(searchQuery));
173 }
174
175 // 2. Apply Type Filter
176 if (activeType !== "all") {
177 filtered = filtered.filter(f => f.type === activeType);
178 }
179
180 // Update counts visually (based on un-searched but type-filtered data)
181 const dirsCount = allFiles.filter(f => f.type === "directory").length;
182 const filesCount = allFiles.filter(f => f.type === "file").length;
183 const fcEl = document.getElementById("folderCount");
184 const flcEl = document.getElementById("fileCount");
185 if (fcEl) fcEl.textContent = dirsCount;
186 if (flcEl) flcEl.textContent = filesCount;
187
188 // 3. Apply Sorting (Directories ALWAYS first, then sort logic)
189 filtered.sort((a, b) => {
190 const isDirA = a.type === "directory" ? 1 : 0;
191 const isDirB = b.type === "directory" ? 1 : 0;
192
193 // Directories always bubble to the top
194 if (isDirA !== isDirB) return isDirB - isDirA;
195
196 if (activeSort === "name-asc") {
197 return a.name.localeCompare(b.name);
198 } else if (activeSort === "name-desc") {
199 return b.name.localeCompare(a.name);
200 } else if (activeSort === "size-desc") {
201 return b.size - a.size;
202 } else if (activeSort === "date-desc") {
203 const tA = new Date(a.modified).getTime();
204 const tB = new Date(b.modified).getTime();
205 return tB - tA;
206 } else if (activeSort === "date-asc") {
207 // Oldest First
208 const tA = new Date(a.modified).getTime() || 0;
209 const tB = new Date(b.modified).getTime() || 0;
210 return tA - tB;
211 }
212 return 0;
213 });
214
215 renderFilesList(filtered, d);
216 }

261 function bindEvents() {
262 const d = typeof window.getDict === 'function' ? window.getDict() : getDict();
263
264 // 1. Search Input Binding
265 const searchInput = document.getElementById("localSearchInput");
266 if (searchInput) {
267 searchInput.addEventListener("input", (e) => {
268 searchQuery = e.target.value.trim().toLowerCase();
269 applyFiltersAndRender();
270 });
271 }
272
273 // 2. Custom Dropdowns Binding
274 const filterWraps = document.querySelectorAll(".filter-dropdown-wrap");
275 filterWraps.forEach(wrap => {
276 const pill = wrap.querySelector(".filter-pill");
277 const valueSpan = pill.querySelector(".filter-value");
278 const options = wrap.querySelectorAll(".filter-option");
279
280 pill.addEventListener("click", (e) => {
281 e.stopPropagation();
282 filterWraps.forEach(w => { if (w !== wrap) w.classList.remove("show"); });
283 wrap.classList.toggle("show");
284 });
285
286 options.forEach(opt => {
287 opt.addEventListener("click", (e) => {
288 e.stopPropagation();
289 const val = opt.getAttribute("data-val");
290 const text = opt.textContent;
291
292 valueSpan.textContent = text;
293 options.forEach(o => o.classList.remove("active"));
294 opt.classList.add("active");
295
296 if (wrap.id === "typeFilterWrap") activeType = val;
297 if (wrap.id === "sortFilterWrap") activeSort = val;
298
299 wrap.classList.remove("show");
300 applyFiltersAndRender();
301 });
302 });
303 });
304
305 // Close dropdowns on outside click
306 document.addEventListener("click", () => {
307 filterWraps.forEach(w => w.classList.remove("show"));
308 });
309
310 // 3. View Toggles Binding
311 const listBtn = document.getElementById("viewList");
312 const gridBtn = document.getElementById("viewGrid");
313 const tableEl = document.getElementById("tableContainer");
314
315 if (listBtn && gridBtn && tableEl) {
316 listBtn.addEventListener("click", () => {
317 tableEl.classList.remove("grid-mode");
318 listBtn.classList.add("active");
319 gridBtn.classList.remove("active");
320 try { localStorage.setItem("view_mode", "list"); } catch { }
321 });
322
323 gridBtn.addEventListener("click", () => {
324 tableEl.classList.add("grid-mode");
325 gridBtn.classList.add("active");
326 listBtn.classList.remove("active");
327 try { localStorage.setItem("view_mode", "grid"); } catch { }
328 });
329
330 // Restore preference
331 try {
332 if (localStorage.getItem("view_mode") === "grid") gridBtn.click();
333 } catch { }
334 }

386 function humanSize(b, lang) {
387 if (!Number.isFinite(b) || b < 0) return "—";
388 if (b === 0) return lang === "fa" ? "0 بایت" : "0 B";
389 const k = 1024;
390 const u = lang === "fa" ? ["بایت", "KB", "MB", "GB", "TB"] : ["B", "KB", "MB", "GB", "TB"];
391 const i = Math.min(Math.floor(Math.log(b) / Math.log(k)), u.length - 1);
392 return (b / Math.pow(k, i)).toFixed(i === 0 ? 0 : 1) + " " + u[i];
393 }
