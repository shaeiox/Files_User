228/* ── Core Language Getters ── */
229function getLang() {
230 try {
231 return localStorage.getItem("ui_lang") || "en";
232 } catch {
233 return "en";
234 }
235}
236
237function getDict(lang) {
238 return i18n[lang || getLang()] || i18n.en;
239}
240
241/* ── Main Language Applicator ── */
242function applyLang(lang) {
243 const targetLang = lang || getLang();
244 const d = getDict(targetLang);
245
246 // 1. Set HTML Direction & Language
247 document.documentElement.lang = d.lang;
248 document.documentElement.dir = d.dir;
249
250 // 2. Standard Text Translation
251 document.querySelectorAll("[data-i18n]").forEach(el => {
252 const k = el.getAttribute("data-i18n");
253 if (d[k] != null) el.textContent = d[k];
254 });
255
256 // 3. Input Placeholders Translation
257 document.querySelectorAll("[data-i18n-ph]").forEach(el => {
258 const k = el.getAttribute("data-i18n-ph");
259 if (d[k] != null) el.placeholder = d[k];
260 });
261
262 // 4. HTML Content Translation (Preserves tags like <span>)
263 document.querySelectorAll("[data-i18n-html]").forEach(el => {
264 const k = el.getAttribute("data-i18n-html");
265 if (d[k] != null) el.innerHTML = d[k];
266 });
267
268 // 5. Update Language Pill Label in Header
269 const currentLangLabel = document.getElementById("currentLangLabel");
270 if (currentLangLabel) {
271 currentLangLabel.textContent = targetLang === "fa" ? "FA" : "EN";
272 }
273
274 // 6. Highlight Active Language in Dropdown
275 document.querySelectorAll(".lang-option").forEach(btn => {
276 if (btn.getAttribute("data-lang") === targetLang) {
277 btn.classList.add("active");
278 } else {
279 btn.classList.remove("active");
280 }
281 });
282
283 // 7. Save Preference
284 try {
285 localStorage.setItem("ui_lang", targetLang);
286 } catch { }
287}
288
289function setupLanguageDropdown() {
290 applyLang(getLang());
291
292 document.addEventListener("click", (e) => {
293 const langOption = e.target.closest(".lang-option");
294 if (langOption) {
295 const selectedLang = langOption.getAttribute("data-lang");
296
297 if (selectedLang && selectedLang !== getLang()) {
298 applyLang(selectedLang);
299
300 if (typeof Router !== "undefined") {
301 Router.refresh();
302 } else {
303 window.location.reload();
304 }
305 }
306 }
307 });
308}
309
310/* ── Auto-Boot on DOM Ready ── */
311if (document.readyState === "loading") {
312 document.addEventListener("DOMContentLoaded", setupLanguageDropdown);
313} else {
314 setupLanguageDropdown();
315}
